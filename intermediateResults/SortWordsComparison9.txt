Please implement an entire C file that takes in the following command line arguments:
-An unspecified number of words (groups of characters separated by whitespace)
The program must:
-Sort these words in ASCII lexicographic order (case-sensitive) order
-Print them to standard output in this sorted order. Write each output word on the same line, with each one separated by a space. End with a newline.

For example, with the command line input:
fruit banana apple carrot grape
The program should print:
apple banana carrot fruit grape

Error handling:
-If no command line arguments are provided, print nothing and exit with returncode 0.
-If any command line argument is not a valid word (that is, a group of characters, it does not have to be a logical English word), print the message: “Error: command line argument not a word” to stderr and exit with returncode 1.
-If any other unexpected error occurs, do not print anything to stderr, and simply exit with returncode 2.
-On success, do not print anything to stderr, and exit with returncode 0.

Please write this code in C. Please write only the code itself, no explanations, markdown fences, or other content, as I want to be able to copy the results directly into a file and run them.

One given solution was: 
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <ctype.h>

int is_valid_word(const char *word) {
    if (word == NULL || *word == '\0') {
        return 0;
    }
    for (const char *c = word; *c; c++) {
        if (!isalpha(*c)) {
            return 0;
        }
    }
    return 1;
}

int compare(const void *a, const void *b) {
    return strcmp(*(const char **)a, *(const char **)b);
}

int main(int argc, char *argv[]) {
    if (argc == 1) {
        return 0;
    }

    char **words = (char **)malloc((argc - 1) * sizeof(char *));
    if (words == NULL) {
        return 2;
    }

    for (int i = 1; i < argc; i++) {
        if (!is_valid_word(argv[i])) {
            fprintf(stderr, "Error: command line argument not a word\n");
            free(words);
            return 1;
        }
        words[i - 1] = argv[i];
    }

    qsort(words, argc - 1, sizeof(char *), compare);

    for (int i = 0; i < argc - 1; i++) {
        printf("%s", words[i]);
        if (i < argc - 2) {
            printf(" ");
        }
    }
    printf("\n");

    free(words);
    return 0;
}
