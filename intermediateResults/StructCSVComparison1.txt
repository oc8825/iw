// Function to split a string by comma and return array of strings
char **split_csv(const char *str, size_t *count) {
    if (!str || !count) return NULL;
    
    // Count commas to determine number of fields
    size_t field_count = 1;
    for (const char *p = str; *p; p++) {
        if (*p == ',') field_count++;
    }
    
    char **fields = (char**)malloc(field_count * sizeof(char*));
    if (!fields) return NULL;
    
    size_t field_index = 0;
    const char *start = str;
    const char *end = str;
    
    while (*end) {
        if (*end == ',' || *(end + 1) == '\0') {
            size_t len = (*end == ',') ? (end - start) : (end - start + 1);
            fields[field_index] = (char*)malloc(len + 1);
            if (!fields[field_index]) {
                // Free previously allocated fields
                for (size_t i = 0; i < field_index; i++) {
                    free(fields[i]);
                }
                free(fields);
                return NULL;
            }
            strncpy(fields[field_index], start, len);
            fields[field_index][len] = '\0';
            field_index++;
            start = end + 1;
        }
        end++;
    }
    
    *count = field_count;
    return fields;
}

// Stable merge sort implementation
void merge_sort_records(struct Record *records, size_t count) {
    if (count <= 1) return;
    
    size_t mid = count / 2;
    struct Record *temp = (struct Record*)malloc(count * sizeof(struct Record));
    if (!temp) exit(2);
    
    merge_sort_records(records, mid);
    merge_sort_records(records + mid, count - mid);
    
    size_t i = 0, j = mid, k = 0;
    
    while (i < mid && j < count) {
        if (compare_records(&records[i], &records[j]) <= 0) {
            temp[k++] = records[i++];
        } else {
            temp[k++] = records[j++];
        }
    }
    
    while (i < mid) temp[k++] = records[i++];
    while (j < count) temp[k++] = records[j++];
    
    for (i = 0; i < count; i++) {
        records[i] = temp[i];
    }
    
    free(temp);
}
